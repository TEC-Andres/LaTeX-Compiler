\begin{frame}{Objetivos del PI 1 y PI 2}
  \orangeHeader{Objetivos del PI 1 y PI 2}
  \vspace{1.5cm}
  \textbf{Objetivos - Problema de investigación 1}
  \begin{itemize}
    \item Conocer cómo afecta la percepción de un consumidor a la hora de elegir un restaurante[cite: 1].
    \item Identificar las preocupaciones del consumidor al elegir un restaurante no reconocido[cite: 1].
  \end{itemize}

  \vspace{0.5cm}
  \textbf{Objetivos - Problema de investigación 2}
  \begin{itemize}
    \item Comprender los valores y elementos conceptuales que generan mayor afinidad e identidad emocional entre los consumidores de hamburguesas y una marca[cite: 1].
    \item Explorar las expectativas sobre experiencias relacionales y dinámicas de interacción (tales como eventos, presencia comunitaria o programas de lealtad) que fomentan el sentido de pertenencia en los clientes[cite: 1].
  \end{itemize}
\end{frame}

\begin{frame}{Objetivos del PI 3 y PI 4}
  \orangeHeader{Objetivos del PI 3 y PI 4}
  \vspace{1.5cm}
  \textbf{Objetivos - Problema de investigación 3}
  \begin{itemize}
    \item Identificar las características que los consumidores valoran en un establecimiento para desarrollar una preferencia por este[cite: 1].
    \item Comprender las preferencias de los consumidores al momento de elegir un establecimiento de comida rápida para comer[cite: 1].
  \end{itemize}

  \vspace{0.5cm}
  \textbf{Objetivos - Problema de investigación 4}
  \begin{itemize}
    \item Determinar los valores en común que la empresa y el consumidor comparten a la hora de hacer uso de sus servicios[cite: 1].
    \item Distinguir ciertas acciones de los consumidores al momento de entrar, consumir y salir del local[cite: 1].
  \end{itemize}
\end{frame}

\begin{frame}{Objetivos del PI 5}
  \orangeHeader{Objetivos del PI 5}
  \vspace{1.5cm}

  \begin{itemize}
    \item Identificar las características demográficas y estilos de vida de los consumidores interesados en smash burgers en Monterrey[cite: 1].
    \item Analizar los hábitos de consumo y las preferencias de compra de clientes potenciales para un lugar de smash burgers[cite: 1].
  \end{itemize}
