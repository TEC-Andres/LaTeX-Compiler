\section{Problemas de decisión gerencial}

\begin{frame}{}
  \orangeHeader{Problemas de Decisión Gerencial}
  \vspace{1.5cm}
  \centering
  \Huge \textbf{SMASH PANDA BURGERS} \\
  \vspace{0.5cm}
  \Large Problemas de Decisión Gerencial (PDG)
\end{frame}

\begin{frame}{PDG 1: Mala ubicación en el mercado}
  \orangeHeader{PDG 1: Mala ubicación en el mercado}
  \vspace{1.5cm}
  \begin{itemize}
    \item No están bien ubicados en el mercado, incluso vendiendo un producto de calidad superior[cite: 1].
    \item La manera en la cual se vende la empresa no transmite confianza[cite: 1].
    \item Existe una desconexión real entre los dueños y su público objetivo[cite: 1].
  \end{itemize}
\end{frame}

\begin{frame}{PDG 2: Falta de información del consumidor}
  \orangeHeader{PDG 2: Falta de información del consumidor}
  \vspace{1.5cm}
  \begin{itemize}
    \item Falta de información sobre las características y preferencias del consumidor objetivo para definir un segmento de mercado prioritario[cite: 1].
  \end{itemize}
\end{frame}

\begin{frame}{PDG 3: Desconocimiento del negocio}
  \orangeHeader{PDG 3: Desconocimiento del negocio}
  \vspace{1.5cm}
  \begin{itemize}
    \item Los consumidores de smash burgers de Monterrey no conocen el negocio[cite: 1].
    \item El único medio de comunicación vigente que Smash Burgers tiene es Instagram y TikTok, teniendo una cobertura baja del establecimiento[cite: 1].
    \item No tiene el reconocimiento de una marca; sus medios de difusión son deficientes[cite: 1].
  \end{itemize}
\end{frame}

\begin{frame}{PDG 4: Retención de clientes}
  \orangeHeader{PDG 4: Retención de clientes}
  \vspace{1.5cm}
  \begin{itemize}
    \item La retención de clientes en un mercado competitivo[cite: 1].
    \item La preferencia del restaurante se basa en darle una experiencia inigualable[cite: 1].
    \item Sin embargo, ellos solo se preocupan en la experiencia inicial[cite: 1].
    \item No hay productos o sistemas para que clientes regulares regresen al establecimiento[cite: 1].
  \end{itemize}
\end{frame}

\begin{frame}{PDG 5 \& 6: Diferenciación y Vínculo Emocional}
  \orangeHeader{PDG 5 \& 6: Diferenciación y Vínculo Emocional}
  \vspace{1.5cm}
  \textbf{PDG 5: No se diferencian}
  \begin{itemize}
    \item Solamente ofreciendo "hamburguesas de calidad" no se diferencian del resto de los competidores los cuales dicen lo mismo[cite: 1].
  \end{itemize}

  \vspace{0.5cm}
  \textbf{PDG 6: Dificultad para generar una relación emocional}
  \begin{itemize}
    \item La empresa no comparte su historia, valores, misión y visión[cite: 1].
    \item Los consumidores no cuentan con la información necesaria para entender el propósito e ideales de la empresa[cite: 1].
  \end{itemize}
\end{frame}
